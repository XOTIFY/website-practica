M1.8
Pușculița — Programare Dinamică
Tehnica: Programare Dinamică
Condiția

Într-o pușculiță se află N monede de diferite valori cu greutatea totală G grame. Greutatea fiecărei monede de o anumită valoare este dată în tabelul ce urmează.

Programul
  using System;

class Program{
    static void Main(){
        int[] valori =   { 1,  5, 10, 25, 50 };
        int[] greutati = { 1,  2,  3,  4,  5 };

        Console.Write("N = ");
        int n = int.Parse(Console.ReadLine());
        Console.Write("G = ");
        int g = int.Parse(Console.ReadLine());

        int[,] tabel = new int[n + 1, g + 1];

        for (int i = 0; i <= n; i++)
            for (int j = 0; j <= g; j++)
                tabel[i, j] = -1;

        tabel[0, 0] = 0;

        for (int i = 1; i <= n; i++){
            for (int j = 0; j <= g; j++){
                for (int t = 0; t < valori.Length; t++){
                    if (j >= greutati[t] && tabel[i - 1, j - greutati[t]] != -1){
                        int suma = tabel[i - 1, j - greutati[t]] + valori[t];
                        if (suma > tabel[i, j])
                            tabel[i, j] = suma;
                    }
                }
            }
        }

        if (tabel[n, g] == -1)
            Console.WriteLine("Nu exista  o combinatie valida pentru N=" + n + " monede si G=" + g + "g");
        else
            Console.WriteLine("Suma maxima este: " + tabel[n, g] + " lei");
    }
}

Al Bundy — Sortare cu Poziție Liberă
Tehnica: Simulare
Condiția

Al Bundy are n+1 cutii de pantofi și n perechi identificate prin valorile 1..n. O cutie este liberă. Dacă scoate o pereche dintr-o cutie, trebuie să o pună imediat în cutia liberă. Ajutați-l să aranjeze pantofii la locurile lor printr-un număr minim de mutări.

Programul
  using System;
using System.IO;

class Program{
    static void Main(){
        string[] linii = File.ReadAllLines("date.in");

        int n = int.Parse(linii[0].Trim());

        string[] initial = linii[1].Trim().Split(' ');
        string[] final2 = linii[2].Trim().Split(' ');

        int[] cutii = new int[n + 2];
        int[] tinta = new int[n + 2];

        for (int i = 1; i <= n + 1; i++)
            cutii[i] = int.Parse(initial[i - 1]);

        for (int i = 1; i <= n + 1; i++)
            tinta[i] = int.Parse(final2[i - 1]);

        int mutari = 0;
        bool gata = false;

        while (!gata){
            int liber = 0;
            for (int i = 1; i <= n + 1; i++)
                if (cutii[i] == 0)
                    liber = i;

            gata = true;
            for (int i = 1; i <= n + 1; i++)
            {
                if (cutii[i] != tinta[i])
                {
                    gata = false;
                    break;
                }
            }

            if (gata) break;

            bool mutat = false;
            for (int i = 1; i <= n + 1; i++){
                if (tinta[liber] != 0 && cutii[i] == tinta[liber]){
                    cutii[liber] = cutii[i];
                    cutii[i] = 0;
                    mutari++;
                    mutat = true;
                    break;
                }
            }

            if (!mutat){
                for (int i = 1; i <= n + 1; i++)
                {
                    if (cutii[i] != 0 && cutii[i] != tinta[i])
                    {
                        cutii[liber] = cutii[i];
                        cutii[i] = 0;
                        mutari++;
                        break;
                    }
                }
            }
        }

        File.WriteAllText("date.out", mutari.ToString());
    }
}

M3.10
Calul de Șah — Backtracking
Tehnica: Metoda Reluării
Condiția

Pe o tablă n×n cu m piese marcate prin -1, un cal pornește din (i0,j0). Să se determine traseul cu număr minim de pași prin care calul mănâncă toate piesele fără a trece de două ori prin aceeași poziție.

Programul
  using System;

class Program
{
	static int n, m;
	static int[,] tabla;
	static int[,] traseu;
	static int[,] traseuBun;
	static int[] dx = { -2, -2, -1, -1, 1, 1, 2, 2 };
	static int[] dy = { -1, 1, -2, 2, -2, 2, -1, 1 };
	static int pasiMinimi;

	static void Main()
	{
		Console.Write("n = ");
		n = int.Parse(Console.ReadLine());

		Console.Write("m = ");
		m = int.Parse(Console.ReadLine());

		tabla = new int[n + 1, n + 1];
		traseu = new int[n + 1, n + 1];
		traseuBun = new int[n + 1, n + 1];

		for (int i = 0; i < m; i++)
		{
			Console.Write("Piesa " + (i + 1) + " (linie coloana): ");
			string[] coords = Console.ReadLine().Split(' ');
			int li = int.Parse(coords[0]);
			int col = int.Parse(coords[1]);
			tabla[li, col] = -1;
		}

		Console.Write("Pozitia calului (linie coloana): ");
		string[] start = Console.ReadLine().Split(' ');
		int i0 = int.Parse(start[0]);
		int j0 = int.Parse(start[1]);

		pasiMinimi = int.MaxValue;

		int mancateStart = 0;
		if (tabla[i0, j0] == -1)
			mancateStart = 1;

		traseu[i0, j0] = 1;
		Backtrack(i0, j0, 1, mancateStart);
		traseu[i0, j0] = 0;

		if (pasiMinimi == int.MaxValue)
		{
			Console.WriteLine("Nu exista traseu valid!");
		}
		else
		{
			Console.WriteLine("Numar minim de pasi: " + (pasiMinimi - 1));
			Console.WriteLine("Traseul:");
			for (int i = 1; i <= n; i++)
			{
				for (int j = 1; j <= n; j++)
					Console.Write(traseuBun[i, j] + "\t");
				Console.WriteLine();
			}
		}
	}

	static void Backtrack(int x, int y, int pas, int mancate)
	{

		if (pas >= pasiMinimi)
			return;


		if (mancate == m)
		{
			if (pas < pasiMinimi)
			{
				pasiMinimi = pas;
				for (int i = 1; i <= n; i++)
					for (int j = 1; j <= n; j++)
						traseuBun[i, j] = traseu[i, j];
			}
			return;
		}

		for (int d = 0; d < 8; d++)
		{
			int nx = x + dx[d];
			int ny = y + dy[d];

			if (nx >= 1 && nx <= n && ny >= 1 && ny <= n && traseu[nx, ny] < 2)
			{
				int mancat = 0;
				if (tabla[nx, ny] == -1 && traseu[nx, ny] == 0)
					mancat = 1;

				traseu[nx, ny]++;
				Backtrack(nx, ny, pas + 1, mancate + mancat);
				traseu[nx, ny]--;
			}
		}
	}
}

M3.25
Puncte Albe și Negre — Backtracking
Tehnica: Metoda Reluării
Condiția

Se dau n puncte albe și n puncte negre în plan. Fiecare punct alb se unește cu câte un punct negru astfel încât oricare două segmente să nu se intersecteze.

Programul
  using System;

class Program
{
    static int n;
    static int[] albeX, albeY;
    static int[] negreX, negreY;
    static int[] pereche;
    static bool[] folosit;

    static void Main()
    {
        Console.Write("n = ");
        n = int.Parse(Console.ReadLine());

        albeX = new int[n];
        albeY = new int[n];
        negreX = new int[n];
        negreY = new int[n];
        pereche = new int[n];
        folosit = new bool[n];

        for (int i = 0; i < n; i++)
        {
            Console.Write("Punct alb " + (i + 1) + " (x y): ");
            string[] coords = Console.ReadLine().Split(' ');
            albeX[i] = int.Parse(coords[0]);
            albeY[i] = int.Parse(coords[1]);
        }

        for (int i = 0; i < n; i++)
        {
            Console.Write("Punct negru " + (i + 1) + " (x y): ");
            string[] coords = Console.ReadLine().Split(' ');
            negreX[i] = int.Parse(coords[0]);
            negreY[i] = int.Parse(coords[1]);
        }

        for (int i = 0; i < n; i++)
            pereche[i] = -1;

        if (Backtrack(0))
        {
            Console.WriteLine("Gasit:");
            for (int i = 0; i < n; i++)
                Console.WriteLine("Punct alb " + (i + 1) + " - Punct negru " + (pereche[i] + 1));
        }
        else
        {
            Console.WriteLine("Nu exista");
        }
    }

    static bool Backtrack(int index)
    {
        if (index == n)
            return true;

        for (int j = 0; j < n; j++)
        {
            if (!folosit[j])
            {
                if (!SeIntersecteaza(index, j))
                {
                    pereche[index] = j;
                    folosit[j] = true;

                    if (Backtrack(index + 1))
                        return true;

                    pereche[index] = -1;
                    folosit[j] = false;
                }
            }
        }

        return false;
    }

    static bool SeIntersecteaza(int alb, int negru)
    {
        for (int i = 0; i < alb; i++)
        {
            int j = pereche[i];
            if (Intersect(albeX[i], albeY[i], negreX[j], negreY[j],
                          albeX[alb], albeY[alb], negreX[negru], negreY[negru]))
                return true;
        }
        return false;
    }

    static long Cross(long ax, long ay, long bx, long by, long cx, long cy){
        return (bx - ax) * (cy - ay) - (by - ay) * (cx - ax);
    }

    static bool Intersect(long ax, long ay, long bx, long by, long cx, long cy, long dx, long dy)
    {
        long d1 = Cross(cx, cy, dx, dy, ax, ay);
        long d2 = Cross(cx, cy, dx, dy, bx, by);
        long d3 = Cross(ax, ay, bx, by, cx, cy);
        long d4 = Cross(ax, ay, bx, by, dx, dy);

        if (d1 * d2 < 0 && d3 * d4 < 0)
            return true;

        return false;
    }
}

M4.5
Generarea Permutărilor — Backtracking
Tehnica: Metoda Reluării
Condiția

Fie dat numărul natural n. Să se genereze toate permutările din n elemente ale mulțimii A={1,2,...,n}.

Programul
  using System;

class Program{
    static int n;
    static int[] perm;
    static bool[] folosit;

    static void Main()
    {
        Console.Write("n = ");
        n = int.Parse(Console.ReadLine());

        perm = new int[n + 1];
        folosit = new bool[n + 1];

        Backtrack(1);
    }

    static void Backtrack(int nivel){
        if (nivel == n + 1){
            for (int i = 1; i <= n; i++)
                Console.Write(perm[i] + " ");
            Console.WriteLine();
            return;
        }

        for (int i = 1; i <= n; i++){
            if (!folosit[i]){
                perm[nivel] = i;
                folosit[i] = true;

                Backtrack(nivel + 1);

                perm[nivel] = 0;
                folosit[i] = false;
            }
        }
    }
}

Minim și Maxim — Divide et Impera
Tehnica: Divide et Impera
Condiția

Scrieți o funcție care determină elementul minim și maxim dintr-un tablou cu elemente întregi folosind metoda divide et impera.

Programul
  using System;

class Program
{
    static int[] v;

    static void Main()
    {
        Console.Write("n = ");
        int n = int.Parse(Console.ReadLine());

        v = new int[n];

        for (int i = 0; i < n; i++)
        {
            Console.Write("v[" + i + "] = ");
            v[i] = int.Parse(Console.ReadLine());
        }

        int min = 0;
        int max = 0;

        MinMax(0, n - 1, ref min, ref max);

        Console.WriteLine("Minim: " + min);
        Console.WriteLine("Maxim: " + max);
    }

    static void MinMax(int st, int dr, ref int min, ref int max){

        if (st == dr){
            min = v[st];
            max = v[st];
            return;
        }

    
        if (dr == st + 1){
            if (v[st] < v[dr])
            {
                min = v[st];
                max = v[dr];
            }
            else
            {
                min = v[dr];
                max = v[st];
            }
            return;
        }

        int mij = (st + dr) / 2;
        int minSt = 0, maxSt = 0;
        int minDr = 0, maxDr = 0;

        MinMax(st, mij, ref minSt, ref maxSt);
        MinMax(mij + 1, dr, ref minDr, ref maxDr);


        if (minSt < minDr)
            min = minSt;
        else
            min = minDr;

        if (maxSt > maxDr)
            max = maxSt;
        else
            max = maxDr;
    }
}

M6.1
Drum Maxim în Matrice — Programare Dinamică
Tehnica: Programare Dinamică
Condiția

Dintr-un element ai,j se poate ajunge în ai+1,j, ai+1,j+1, ai+1,j-1. Să se găsească drumul de la linia 1 la linia n cu suma maximă.

Programul
  using System;

class Program{
    static void Main(){
        Console.Write("n = ");
        int n = int.Parse(Console.ReadLine());
        int[,] a = new int[n + 1, n + 1];
        int[,] dp = new int[n + 1, n + 1];

        for (int i = 1; i <= n; i++){
            for (int j = 1; j <= n; j++){
                Console.Write("a[" + i + "][" + j + "] = ");
                a[i, j] = int.Parse(Console.ReadLine());
            }
        }

        for (int j = 1; j <= n; j++)
            dp[1, j] = a[1, j];

        for (int i = 2; i <= n; i++){
            for (int j = 1; j <= n; j++){
                int maxPrecedent = dp[i - 1, j]; 

                if (j > 1 && dp[i - 1, j - 1] > maxPrecedent)
                    maxPrecedent = dp[i - 1, j - 1]; 

                if (j < n && dp[i - 1, j + 1] > maxPrecedent)
                    maxPrecedent = dp[i - 1, j + 1]; 

                dp[i, j] = a[i, j] + maxPrecedent;
            }
        }

        int sumaMax = dp[n, 1];
        int colMax = 1;
        for (int j = 2; j <= n; j++)
        {
            if (dp[n, j] > sumaMax)
            {
                sumaMax = dp[n, j];
                colMax = j;
            }
        }

        int[] drum = new int[n + 1];
        drum[n] = colMax;

        for (int i = n - 1; i >= 1; i--)
        {
            int j = drum[i + 1];
            int best = j;

            if (j > 1 && dp[i, j - 1] > dp[i, best])
                best = j - 1;

            if (j < n && dp[i, j + 1] > dp[i, best])
                best = j + 1;

            drum[i] = best;
        }

        Console.WriteLine("Suma maxima: " + sumaMax);
        Console.Write("Drumul: ");
        for (int i = 1; i <= n; i++)
            Console.Write("a[" + i + "," + drum[i] + "] ");
        Console.WriteLine();
    }
}

M6.22
Suma în Triunghi — Programare Dinamică
Tehnica: Programare Dinamică
Condiția

Fie un triunghi format din n linii, fiecare linie conținând numere întregi din [1,99]. Să se determine suma maximă de la vârf la bază.

Programul
  using System;

class Program
{
    static void Main()
    {
        Console.Write("n = ");
        int n = int.Parse(Console.ReadLine());

        int[,] triunghi = new int[n + 1, n + 1];
        int[,] dp = new int[n + 1, n + 1];

        for (int i = 1; i <= n; i++)
        {
            Console.Write("Linia " + i + " (" + i + " numere): ");
            string[] vals = Console.ReadLine().Split(' ');
            for (int j = 1; j <= i; j++)
                triunghi[i, j] = int.Parse(vals[j - 1]);
        }

 
        dp[1, 1] = triunghi[1, 1];

        for (int i = 2; i <= n; i++){
            for (int j = 1; j <= i; j++){
                int maxPrecedent = 0;

         
                if (j > 1)
                    maxPrecedent = dp[i - 1, j - 1];

       
                if (j <= i - 1 && dp[i - 1, j] > maxPrecedent)
                    maxPrecedent = dp[i - 1, j];

                dp[i, j] = triunghi[i, j] + maxPrecedent;
            }
        }

        int sumaMax = dp[n, 1];
        int colMax = 1;
        for (int j = 2; j <= n; j++){
            if (dp[n, j] > sumaMax)
            {
                sumaMax = dp[n, j];
                colMax = j;
            }
        }

        int[] drum = new int[n + 1];
        drum[n] = colMax;

        for (int i = n - 1; i >= 1; i--)
        {
            int j = drum[i + 1];

            if (j > 1 && dp[i, j - 1] > dp[i, j])
                drum[i] = j - 1;
            else
                drum[i] = j;
        }

        Console.WriteLine("Suma maxima: " + sumaMax);
        Console.Write("Drumul: ");
        for (int i = 1; i <= n; i++)
            Console.Write(triunghi[i, drum[i]] + " ");
        Console.WriteLine();
    }
}

Gestiunea CD-urilor — POO
Tehnica: Programare Orientată pe Obiecte
Condiția

Să se dezvolte o aplicație care ține evidența CD-urilor: introducere, afișare sortată după nume, adăugare, căutare după ID, filtrare după spațiu liber, ordonare după dimensiune, afișare sesiuni deschise, clasare după viteză.

Programul
  using System;
using System.Collections.Generic;

class CD{
    public int Id;
    public string Nume;
    public double Dimensiune;
    public int VitezaMaxima;
    public double SpatiuOcupat;
    public int NrSesiuni;
    public string TipSesiune;

    public double SpatiuLiber()
    {
        return Dimensiune - SpatiuOcupat;
    }
}

class Program{
    static List lista = new List();

    static void Main()
    {
        int optiune = 0;
        while (optiune != 9)
        {
            Console.WriteLine("\n GESTIUNE CD :");
            Console.WriteLine("1. Introducere date");
            Console.WriteLine("2. Afisare CD-uri sortate dupa nume");
            Console.WriteLine("3. Adaugare CD nou");
            Console.WriteLine("4. Cautare CD dupa identificator");
            Console.WriteLine("5. Afisare CD-uri cu spatiu liber mai mare decat o valoare");
            Console.WriteLine("6. Ordonare CD-uri dupa dimensiune");
            Console.WriteLine("7. Afisare CD-uri cu sesiuni deschise");
            Console.WriteLine("8. Clasare CD-uri dupa viteza de inscriptionare");
            Console.WriteLine("9. Iesire");
            Console.Write("Optiune: ");
            optiune = int.Parse(Console.ReadLine());

            if (optiune == 1)
                IntroducereDate();
            else if (optiune == 2)
                AfisareSortateDupaNume();
            else if (optiune == 3)
                AdaugareCD();
            else if (optiune == 4)
                CautareDupaId();
            else if (optiune == 5)
                AfisareSpatiuLiber();
            else if (optiune == 6)
                OrdonareDupaDimensiune();
            else if (optiune == 7)
                AfisareSesiuniDeschise();
            else if (optiune == 8)
                ClasareDupaViteza();
            else if (optiune == 9)
                Console.WriteLine("La revedere!");
            else
                Console.WriteLine("Optiune invalida!");
        }
    }

    static CD CitireCD()
    {
        CD cd = new CD();
        Console.Write("Id: ");
        cd.Id = int.Parse(Console.ReadLine());
        Console.Write("Nume: ");
        cd.Nume = Console.ReadLine();
        Console.Write("Dimensiune (MB): ");
        cd.Dimensiune = double.Parse(Console.ReadLine());
        Console.Write("Viteza maxima de inscriptionare: ");
        cd.VitezaMaxima = int.Parse(Console.ReadLine());
        Console.Write("Spatiu ocupat (MB): ");
        cd.SpatiuOcupat = double.Parse(Console.ReadLine());
        Console.Write("Numar sesiuni: ");
        cd.NrSesiuni = int.Parse(Console.ReadLine());
        Console.Write("Tip sesiune (deschisa/inchisa): ");
        cd.TipSesiune = Console.ReadLine();
        return cd;
    }

    static void AfisareCD(CD cd)
    {
        Console.WriteLine("----------------------------");
        Console.WriteLine("Id: " + cd.Id);
        Console.WriteLine("Nume: " + cd.Nume);
        Console.WriteLine("Dimensiune: " + cd.Dimensiune + " MB");
        Console.WriteLine("Viteza maxima: " + cd.VitezaMaxima + "x");
        Console.WriteLine("Spatiu ocupat: " + cd.SpatiuOcupat + " MB");
        Console.WriteLine("Spatiu liber: " + cd.SpatiuLiber() + " MB");
        Console.WriteLine("Numar sesiuni: " + cd.NrSesiuni);
        Console.WriteLine("Tip sesiune: " + cd.TipSesiune);
    }

    static void IntroducereDate()
    {
        lista.Clear();
        Console.Write("Cate CD-uri introduceti :  ");
        int n = int.Parse(Console.ReadLine());
        for (int i = 0; i < n; i++)
        {
            Console.WriteLine("\nCD " + (i + 1) + ":");
            lista.Add(CitireCD());
        }
        Console.WriteLine("Date introduse cu succes!");
    }

    static void AdaugareCD()
    {
        Console.WriteLine("\nDate CD nou:");
        lista.Add(CitireCD());
        Console.WriteLine("CD adaugat cu succes!");
    }

    static void AfisareSortateDupaNume()
    {
        if (lista.Count == 0)
        {
            Console.WriteLine("Nu exista CD-uri!");

class Program
{
    static void Main()
    {
        List persoane = new List();

        while (true)
        {
            Console.WriteLine("\n==============================");
            Console.WriteLine("1. Adauga Student");
            Console.WriteLine("2. Adauga Angajat");
            Console.WriteLine("3. Adauga Student-Angajat");
            Console.WriteLine("4. Adauga Administrator");
            Console.WriteLine("5. Afiseaza toate persoanele");
            Console.WriteLine("6. Exclude persoana");
            Console.WriteLine("7. Total salarii");
            Console.WriteLine("8. Total burse");
            Console.WriteLine("9. Total general");
            Console.WriteLine("0. Iesire");
            Console.WriteLine("==============================");
            Console.Write("Optiunea: ");

            int opt;
            if (!int.TryParse(Console.ReadLine(), out opt))
                continue;

            Persoana p = null;

            switch (opt)
            {
                case 1:
                    p = new Student();
                    p.Citire();
                    persoane.Add(p);
                    break;

                case 2:
                    p = new Angajat();
                    p.Citire();
                    persoane.Add(p);
                    break;

                case 3:
                    p = new StudentAngajat();
                    p.Citire();
                    persoane.Add(p);
                    break;

                case 4:
                    p = new Administrator();
                    p.Citire();
                    persoane.Add(p);
                    break;

                case 5:
                    if (persoane.Count == 0)
                    {
                        Console.WriteLine("Nu exista persoane.");
                    }
                    else
                    {
                        foreach (Persoana persoana in persoane)
                        {
                            Console.WriteLine("\nTip: " + persoana.Tip());
                            persoana.Afisare();
                        }
                    }
                    break;

                case 6:
                    Console.Write("Introdu numele persoanei de sters: ");
                    string nume = Console.ReadLine();

                    bool gasit = false;

                    for (int i = 0; i < persoane.Count; i++)
                    {
                        if (persoane[i].GetNume().ToLower() == nume.ToLower())
                        {
                            persoane.RemoveAt(i);
                            gasit = true;
                            break;
                        }
                    }

                    if (gasit == true)
{
    Console.WriteLine("Persoana eliminata.");
}
else
{
    Console.WriteLine("Persoana nu a fost gasita.");
}
break;
                case 7:
                    double totalSalarii = 0;

                    foreach (Persoana persoana in persoane)
                        totalSalarii += persoana.Salariu();

                    Console.WriteLine($"Total salarii = {totalSalarii}");
                    break;

                case 8:
                    double totalBurse = 0;

                    foreach (Persoana persoana in persoane)
                        totalBurse += persoana.Bursa();

                    Console.WriteLine($"Total burse = {totalBurse}");
                    break;

                case 9:
                    double totalGeneral = 0;

                    foreach (Persoana persoana in persoane)
                        totalGeneral += persoana.Salariu() + persoana.Bursa();

                    Console.WriteLine($"Total general = {totalGeneral}");
                    break;

                case 0:
                    return;

                default:
                    Console.WriteLine("Optiune invalida!");
                    break;
            }
        }
    }
}
